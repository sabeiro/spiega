#include <Arduino.h>
#include <WiFi.h>
#include <HTTP.h>

void setup() {
  Serial.begin(115200);
  Serial.println("M5Stack Core 2 Booting...");
  
  delay(1000);
  
  WiFi.begin("IoT_NETWORK", "iot_passphrase");
  
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
  }
  
  Serial.println("WiFi Connected!");
  Serial.println(WiFi.localIP());
  
  // Subscribe to MQTT topic
  subscribe_mqtt_topic();
  
  // Initialize sensors
  initialize_sensors();
}

void loop() {
  read_sensors();
  publish_sensor_data();
  
  delay(100);
}

** MQTT Configuration
```cpp
// MQTT setup for Core 2
const char* mqtt_server = "mqtt.blendertwin.local";
const char* topic = "m5stack/core2/data";

#include <ArduinoJson.h>

void publish_sensor_data() {
  DynamicJsonDocument doc(1024);
  
  doc["temperature"] = read_temperature();
  doc["humidity"] = read_humidity();
  doc["battery"] = get_battery_percent();
  doc["voltage"] = read_voltage();
  
  String json = String(doc);
  mqtt_publish(topic, json);
}
```
