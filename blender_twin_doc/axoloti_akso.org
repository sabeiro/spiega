* Code Examples

** axoloti Audio Loop
#+BEGIN_SRC cpp
#include <axoloti_core.h>
#include <axoloti_dsp.h>

// Initialize audio
void axoloti_init() {
  // Setup audio buffer
  audio_buffer.allocate(4096);
  audio_buffer.fill_with_silence();
}

// Audio processing loop
void axoloti_loop() {
  // Read input
  int16_t sample = audio_input.read();
  
  // Apply DSP
  float gain = 0.8f;
  float filtered = sample * gain;
  
  // Write to output
  audio_output.write((int16_t)filtered);
}

// Audio initialization
void axoloti_start() {
  axoloti_init();
  axoloti_loop();
}

** akso MQTT Bridge
#+BEGIN_SRC cpp
#include <PubSubClient.h>
#include <axoloti_mqtt.h>

// MQTT connection to Blender Twin
PubSubClient client(server, 1883, callback);

void setup() {
  Serial.begin(115200);
  client.begin("axoloti.local");
  client.setServer("broker.blendertwin.local", 1883);
  
  // Subscribe to topics
  client.subscribe("axoloti/control/");
}

void loop() {
  client.loop();
  
  // Publish status
  client.publish("axoloti/status", "active");
  delay(1000);
}
