#+name: docker_install
#+begin_src bash
sudo apt update
sudo apt install ca-certificates curl
sudo install -m 0755 -d /etc/apt/keyrings
sudo curl -fsSL https://download.docker.com/linux/ubuntu/gpg -o /etc/apt/keyrings/docker.asc
sudo chmod a+r /etc/apt/keyrings/docker.asc
sudo tee /etc/apt/sources.list.d/docker.sources <<EOF
Types: deb
URIs: https://download.docker.com/linux/ubuntu
Suites: $(. /etc/os-release && echo "${UBUNTU_CODENAME:-$VERSION_CODENAME}")
Components: stable
Architectures: $(dpkg --print-architecture)
Signed-By: /etc/apt/keyrings/docker.asc
EOF
sudo apt update
sudo apt install docker-ce docker-ce-cli containerd.io docker-buildx-plugin docker-compose-plugin
sudo systemctl status docker
sudo systemctl start docker
sudo groupadd docker
sudo usermod -aG docker $USER
newgrp docker
#+end_src

* docker 

Here are useful Docker commands for working with =.org= (Org-mode) files:

*Build & Dev:*
- =docker build -f Dockerfile.dockerfile -t myorg .= - Build from Dockerfile
- =docker run -it --rm -v $PWD:/workspace myorg org mode :start= - Launch org-mode
- =docker exec -it myorg bash= - Enter container
- =docker build -f Dockerfile.dockerfile --build-arg DOCKER_FLAGS="-v $PWD:/workspace" -t myorg .= - Build with volume flags

*Run & Manage:*
- =docker run -it --rm -v $PWD:/workspace -w /workspace myorg org mode --no-user-info= - Run org without user info
- =docker run --rm -v $PWD:/workspace myorg org mode :source_block_quote: "hello"= - Process org block
- =docker run --rm myorg org mode :clear_properties= - Clear properties
- =docker run --rm myorg org mode :close_all= - Close all buffers

*Network & Logs:*
- =docker run --rm -v $PWD:/workspace myorg org mode :start_export "latex"= - Export to LaTeX
- =docker logs myorg= - View logs
- =docker ps -a= - List all containers

*Useful for Org workflows:*
- =docker cp myorg:/workspace/src/*.org $PWD/= - Copy files out
- =docker cp $PWD/*.org myorg:/workspace/src/= - Copy files in
- =docker commit myorg myorg:latest= - Save image
- =docker stop myorg && docker rm myorg= - Stop and remove
