*** video to gif

We need to convert video clips into light gif for blog posts

#+name: video_to_gif
#+begin_src bash :exports both :results output replace
cd ~/VideoProd/clip/
for i in *mp4; do
    echo ${i/.mp4/.gif} 
    #ffmpeg -i $i -y -vf "fps=10,scale:500-1,palettegen" -loop 0 ${i/.mp4/.gif}
    ffmpeg -i $i -vf "palettegen" -y palette.png
    #ffmpeg -i $i -y -vf "fps=10" -i palette.png -filter_complex "paletteuse" -loop 0 ${i/.mp4/.gif}
    ffmpeg -i $i -y -vf "fps=10,scale=500:-1:flags=lanczos,split[s0][s1];[s0]palettegen=max_colors=32:stats_mode=single:[p];[s1][p]paletteuse=dither=floyd_steinberg" \
	   ${i/.mp4/.gif}
done
#+end_src
